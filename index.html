📋 Google Sheets 串接設定說明

步驟一：建立試算表欄位

在 Google 試算表新增工作表，命名為「清運紀錄」，第一列填入以下標題（順序不可改）：

A1: 日期
B1: 板數
C1: 可回收重量(kg)
D1: 可回收金額(元)
E1: 不可回收重量(kg)
F1: 不可回收金額(元)
G1: 總金額(元)
H1: 報廢物說明
I1: 照片連結
J1: 影片連結
K1: 建立時間

步驟二：貼入 Apps Script 程式碼

試算表 → 擴充功能 → Apps Script，貼入以下內容後儲存：

const SHEET_NAME = '清運紀錄';

function doPost(e) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet()
                    .getSheetByName(SHEET_NAME);
    const raw = e.postData ? e.postData.contents : e.parameter.data;
    const d = JSON.parse(raw);
    const row = [
      d.date, d.boards,
      d.recKg, d.recAmt,
      d.nonKg, d.nonAmt,
      d.totalAmt,
      d.desc,
      d.photoUrl || '',
      d.videoUrl || '',
      new Date().toLocaleString('zh-TW')
    ];
    if (d.action === 'update' && d.rowIndex) {
      // 修改：覆寫指定列
      const lastCol = sheet.getLastColumn();
      sheet.getRange(d.rowIndex, 1, 1, lastCol).clearContent();
      sheet.getRange(d.rowIndex, 1, 1, row.length).setValues([row]);
    } else {
      // 新增
      sheet.appendRow(row);
    }
    return ok({status:'saved'});
  } catch(err) {
    return ok({status:'error', message: err.message});
  }
}

function doGet() {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet()
                    .getSheetByName(SHEET_NAME);
    const rows = sheet.getDataRange().getValues();
    rows.shift();
    const data = rows.map(r => {
      let rawDate = r[0];
      let dateStr = '';
      if (rawDate instanceof Date) {
        dateStr = `${rawDate.getFullYear()}-${String(rawDate.getMonth()+1).padStart(2,'0')}-${String(rawDate.getDate()).padStart(2,'0')}`;
      } else {
        dateStr = String(rawDate).slice(0,10);
      }
      return {
        date: dateStr,
        boards: Number(r[1]),
        recKg: Number(r[2]),  recAmt: Number(r[3]),
        nonKg: Number(r[4]),  nonAmt: Number(r[5]),
        totalAmt: Number(r[6]),
        desc: r[7],
        photoUrl: r[8] || '',
        videoUrl: r[9] || ''
      };
    });
    return ok(data);
  } catch(err) {
    return ok({status:'error', message: err.message});
  }
}

function ok(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

步驟三：部署為網頁應用程式

  1. 點右上角「部署 → 新增部署」
  2. 類型選「網頁應用程式」
  3. 執行身分：「我」；存取：「所有人」
  4. 點「部署」，複製產生的網址

步驟四：填入網址

用文字編輯器打開本 HTML 檔，找到開頭的 SCRIPT_URL，替換成剛複製的網址，存檔即完成。
