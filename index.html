const SHEET_NAME = '清運紀錄'; function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet() .getSheetByName(SHEET_NAME); const raw = e.postData ? e.postData.contents : e.parameter.data; const d = JSON.parse(raw); sheet.appendRow([ d.date, d.boards, d.recKg, d.recAmt, d.nonKg, d.nonAmt, d.totalAmt, d.desc, d.driveUrl || '', new Date().toLocaleString('zh-TW') ]); return ok({status:'saved'}); } catch(err) { return ok({status:'error', message: err.message}); } } function doGet(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet() .getSheetByName(SHEET_NAME); const rows = sheet.getDataRange().getValues(); rows.shift(); const data = rows.map(r => ({ date: String(r[0]).slice(0,10), boards: Number(r[1]), recKg: Number(r[2]), recAmt: Number(r[3]), nonKg: Number(r[4]), nonAmt: Number(r[5]), totalAmt: Number(r[6]), desc: r[7], driveUrl: r[8] || '' })); return ok(data); } catch(err) { return ok({status:'error', message: err.message}); } } function ok(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
